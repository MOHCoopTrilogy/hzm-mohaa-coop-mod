setup
{
	scale 0.6	// Needs to be unnaturally large so it is visible
	path models/projectiles/M18_Smoke_grenade
	skelmodel M18_Smoke_grenade.skd
	surface all shader M18_smoke_gren
}


init
{
	server
	{
		classname Projectile
		anim idle
		setsize       "-2 -2 -2" "2 2 2"
		speed         700
		minspeed      300
		life			4.0
		dmlife			3.0
		hitdamage     10
		health        50
		takedamage
		meansofdeath		grenade

		explosionmodel models/fx/M18_Smoke_grenadeExplosion.tik
		bouncetouch
		bouncesound		smokegrenade_bounce_soft
		bouncesound_metal	smokegrenade_bounce_metal
		bouncesound_hard	smokegrenade_bounce_hard
		bouncesound_water	smokegrenade_bounce_water
		smashthroughglass 64
		gravity 0.8
		avelocity random 90 random 90 60
		dieinwater

		cache models/fx/M18_Smoke_grenadeExplosion.tik
	}

	client
	{
		// HZM coop: in-flight trail SUPPRESSED (spawnrate 0). The red smokegrenade_allied
		// sprite is replaced by the yellow drop-zone plume spawned on landing.
		originemitter smoke
		(
			spawnrate 0
			model smokegrenade_allied.spr
			color 1 1 1
			spritegridlighting
			alpha .85
			bouncefactor 0.8
			life 2.25 3.5
			scalerate .8
			velocity 24
			randvelaxis crandom 4 crandom 4 4
			spin 0.133
			angles 0 0 random 360
			avelocity 0 0 crandom 20
			scalemin 0.1
			scalemax 0.25
			fadedelay 2
			fadein 0.25
		)
	}
}

animations
{
	idle	M18_Smoke_grenade.skc
	{
		client
		{
			entry loopsound smokegrenade_air_loop
		}
	}
}
